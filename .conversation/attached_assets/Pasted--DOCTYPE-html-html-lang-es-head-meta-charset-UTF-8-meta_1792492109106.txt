<script>
  let precio = 20000;
  let inicial = 2000;
  let meses = 24;
  let tasaMensual = 0; // en porcentaje

  function setPrecio(valor) {
    precio = valor;
    document.getElementById('precioInput').value = valor;
    document.querySelectorAll('.opt-btn').forEach(b => {
      if (b.innerText.includes(valor.toLocaleString())) b.classList.add('active');
      else if (b.innerText.includes('15,000') || b.innerText.includes('20,000') || b.innerText.includes('25,000')) {
        b.classList.remove('active');
      }
    });
    setInicialPct(10);
  }

  function manualPrecio() {
    precio = parseFloat(document.getElementById('precioInput').value) || 0;
    actualizarMaxInicial();
    actualizar();
  }

  function setInicialPct(pct) {
    inicial = Math.round(precio * (pct / 100));
    document.getElementById('inicialInput').value = inicial;
    document.getElementById('inicialSlider').value = inicial;
    actualizar();
  }

  function sliderInicial() {
    inicial = parseFloat(document.getElementById('inicialSlider').value);
    document.getElementById('inicialInput').value = inicial;
    actualizar();
  }

  function manualInicial() {
    inicial = parseFloat(document.getElementById('inicialInput').value) || 0;
    document.getElementById('inicialSlider').value = inicial;
    actualizar();
  }

  function actualizarMaxInicial() {
    const s = document.getElementById('inicialSlider');
    s.max = precio;
    if (inicial > precio) inicial = precio;
  }

  function setPlazo(cantMeses, btn) {
    meses = cantMeses;
    document.querySelectorAll('.plazo-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    actualizar();
  }

  function calcularCuota(capital, tasa, n) {
    if (tasa === 0) return capital / n;
    let r = tasa / 100;
    return capital * (r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);
  }

  function actualizar() {
    actualizarMaxInicial();
    tasaMensual = parseFloat(document.getElementById('interesSlider').value);
    document.getElementById('txtInteres').innerText = tasaMensual + '%';

    let pctInicial = precio > 0 ? Math.round((inicial / precio) * 100) : 0;
    document.getElementById('txtInicialPct').innerText = pctInicial + '%';

    let saldo = Math.max(0, precio - inicial);
    let cuota = saldo > 0 ? calcularCuota(saldo, tasaMensual, meses) : 0;
    let totalFinanciadoConInteres = cuota * meses;
    let totalIntereses = Math.max(0, totalFinanciadoConInteres - saldo);
    let totalFinal = inicial + totalFinanciadoConInteres;

    // Pintar valores
    document.getElementById('resCuota').innerText = 'S/ ' + cuota.toFixed(2);
    document.getElementById('resSaldo').innerText = 'S/ ' + saldo.toFixed(2);
    document.getElementById('resInteresTotal').innerText = 'S/ ' + totalIntereses.toFixed(2);
    document.getElementById('resPlazoTexto').innerText = meses + ' Meses';
    document.getElementById('resTotalPagar').innerText = 'S/ ' + totalFinal.toFixed(2);
  }

  function compartirWhatsApp() {
    let cuota = document.getElementById('resCuota').innerText;
    let saldo = document.getElementById('resSaldo').innerText;
    let texto = `*🏡 COTIZACIÓN DE LOTE*%0A` +
                `----------------------------------%0A` +
                `*Precio Total:* S/ ${precio.toLocaleString()}%0A` +
                `*Inicial:* S/ ${inicial.toLocaleString()} (${document.getElementById('txtInicialPct').innerText})%0A` +
                `*Saldo a financiar:* ${saldo}%0A` +
                `*Plazo:* ${meses} meses%0A` +
                `*Tasa mensual:* ${tasaMensual}%%0A` +
                `*Cuota mensual:* ${cuota}%0A` +
                `----------------------------------%0A` +
                `_¿Deseas agendar una visita al terreno?_`;
    window.open(`https://api.whatsapp.com/send?text=${texto}`, '_blank');
  }

  // Inicializar
  actualizar();
</script>
